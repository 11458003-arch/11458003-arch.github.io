// 單一檔案 Google Apps Script - 雲端記事本 const FOLDER_ID = '1I3k3jWlV5XfVT-La-J5ewmbJ_JN-WcBY'; const FILE_NAME = '記事本.txt'; function doGet(e) { const html = ` 雲端記事本
☁️

雲端記事本

檔案：${FILE_NAME}
載入中…
`; return HtmlService.createHtmlOutput(html).setTitle('雲端記事本'); } // 取得或建立檔案 function _getFile() { const folder = DriveApp.getFolderById(FOLDER_ID); const files = folder.getFilesByName(FILE_NAME); if (files.hasNext()) return files.next(); return folder.createFile(FILE_NAME, ''); } // 供前端載入內容與最後修改時間 function getContent() { const file = _getFile(); const blob = file.getBlob(); const text = blob.getDataAsString(); const lastUpdated = file.getLastUpdated(); return { content: text, lastUpdated: lastUpdated ? lastUpdated.toISOString() : '' }; } // 儲存內容（使用鎖避免競爭） function saveContent(content) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const file = _getFile(); file.setContent(content); const last = file.getLastUpdated(); return { success: true, lastUpdated: last ? last.toISOString() : '' }; } finally { try { lock.releaseLock(); } catch(e){} } }
